# Development only. Never installed on the robot.
#
# The emulator and the face preview both need Pillow and tkinter. tkinter ships
# with python.org and Windows builds; on Debian it is python3-tk from apt.
-r ../firmware/requirements.txt

# Real microphone and speaker in the emulator, through PortAudio. Optional:
# without it the audio backends refuse with an install line and the level
# slider keeps working, so the rest of the emulator is unaffected.
sounddevice>=0.4

# wakeword/ui's phone-recording page needs an HTTPS dev server -- phone
# browsers refuse microphone access over plain http:// to any address other
# than localhost, so the dashboard serves itself over a self-signed cert
# (Flask's `ssl_context="adhoc"`, which needs this package).
cryptography>=42

# The scheduler keeps reminders in the owner's timezone with zoneinfo. The robot
# and CI have the system tz database; Windows has none, and Python there only
# gets one from this package. Development only: the firmware does not install
# it, and it is not in the release bill of materials.
tzdata>=2024.1

# The wake word and the end-of-speech detector run on the PC emulator too, so
# the whole product can be tried on a laptop (docs/plans/09-build-order.md).
# The firmware's own list for the board is firmware/requirements-pi.txt.
pymicro-wakeword>=2.0
